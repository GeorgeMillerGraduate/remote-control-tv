cmake_minimum_required(VERSION 3.24)
project(JengaRemoteCpp VERSION 0.2.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
if(WIN32)
  enable_language(RC)
endif()
find_package(Qt6 6.4 REQUIRED COMPONENTS Widgets Network WebSockets Concurrent Test)
find_package(OpenSSL 3 REQUIRED)
add_library(remote_core src/security.cpp src/security.h src/input.cpp src/input.h src/server.cpp src/server.h third_party/qrcodegen.cpp)
target_include_directories(remote_core PUBLIC src third_party)
target_compile_options(remote_core PUBLIC $<$<CXX_COMPILER_ID:MSVC>:/utf-8>)
target_link_libraries(remote_core PUBLIC Qt6::Widgets Qt6::Network Qt6::WebSockets Qt6::Concurrent OpenSSL::Crypto)
if(WIN32)
  target_compile_definitions(remote_core PUBLIC NOMINMAX WIN32_LEAN_AND_MEAN UNICODE _UNICODE _WIN32_WINNT=0x0A00)
  target_link_libraries(remote_core PUBLIC user32 gdi32 advapi32 crypt32)
else()
  find_package(X11 REQUIRED)
  find_library(XTST_LIBRARY NAMES Xtst libXtst.so.6 REQUIRED)
  target_link_libraries(remote_core PUBLIC X11::X11 ${XTST_LIBRARY})
endif()
qt_add_executable(JengaRemote src/main.cpp)
target_link_libraries(JengaRemote PRIVATE remote_core)
if(WIN32)
  set_target_properties(JengaRemote PROPERTIES WIN32_EXECUTABLE TRUE)
  target_sources(JengaRemote PRIVATE assets/windows.rc assets/windows.manifest)
  target_include_directories(JengaRemote PRIVATE assets)
endif()
include(CTest)
if(BUILD_TESTING)
  qt_add_executable(remote_tests tests/test_remote.cpp)
  target_link_libraries(remote_tests PRIVATE remote_core Qt6::Test)
  add_test(NAME remote_tests COMMAND remote_tests)
  set_tests_properties(remote_tests PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen")
endif()
install(TARGETS JengaRemote RUNTIME DESTINATION .)
